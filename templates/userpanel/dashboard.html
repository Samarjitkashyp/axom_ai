Sovereign AI Integration Quickstart

Connect to Axom AI in Under 2 Minutes

Sovereign Assamese AI engines: Xuna (STT), Puhor (TTS) & Xetu (Translation) with 100% OpenAI SDK compatibility.

from openai import OpenAI

# Point official OpenAI SDK to Axom AI (Zero code changes)
client = OpenAI(
    api_key="YOUR_AXOM_API_KEY",
    base_url="https://aiaxom.co.in/v1"
)

# Xuna 600M Conformer ASR (Pure Assamese script ৰ/ৱ)
audio_file = open("speech.mp3", "rb")
transcription = client.audio.transcriptions.create(
    model="axomai-speech-xuna",
    file=audio_file,
    language="as"
)

print(transcription.text)

import requests

# Xetu 200M Transformer Neural Translation Bridge (en <-> as)
url = "https://aiaxom.co.in/v1/translate"
headers = {
    "Authorization": "Bearer YOUR_AXOM_API_KEY",
    "Content-Type": "application/json"
}
payload = {
    "text": "Artificial Intelligence transforms local education in Assam.",
    "source_lang": "en",
    "target_lang": "as",
    "model": "axomai-trans-xetu"
}

response = requests.post(url, json=payload, headers=headers)
print(response.json()["translated_text"])

from openai import OpenAI

client = OpenAI(
    base_url="https://aiaxom.co.in/v1",
    api_key="YOUR_AXOM_SECRET_KEY"
)

# Transcribe speech with Sovereign Xuna ASR

        file=audio_file,
        language="as"

print(res.text)
